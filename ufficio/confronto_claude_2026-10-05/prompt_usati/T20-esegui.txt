'use strict';

/* ========== 1. FUNZIONI PURE ========== */

/** Livelli selezionabili nel filtro della home. */
const LIVELLI = ['A1', 'A2', 'tutti'];

/** Chiavi dei tre giochi (sono anche le chiavi dei dati in window.LANGUE_DATA). */
const GIOCHI = ['articoli', 'frasi', 'errori'];

/** Mescola una copia della lista (Fisher-Yates). `rng` è una funzione che dà un numero in [0,1). */
function mescola(lista, rng) {
  const casuale = rng || Math.random;
  const copia = lista.slice();
  for (let i = copia.length - 1; i > 0; i--) {
    const j = Math.floor(casuale() * (i + 1));
    const tmp = copia[i];
    copia[i] = copia[j];
    copia[j] = tmp;
  }
  return copia;
}

/** Sceglie un elemento a caso, evitando se possibile quello appena uscito (`escludi`). Lista vuota -> null. */
function scegliCasuale(lista, rng, escludi) {
  if (!lista || lista.length === 0) return null;
  const casuale = rng || Math.random;
  let candidati = lista.filter(function (x) { return x !== escludi; });
  if (candidati.length === 0) candidati = lista;
  return candidati[Math.floor(casuale() * candidati.length)];
}

/** Un record è usabile nel sito solo se non è "da_verificare". */
function eUsabile(record) {
  return !!record && record.stato !== 'da_verificare';
}

/** Tiene i record usabili del livello richiesto ("tutti" = nessun filtro sul livello). */
function filtraPerLivello(lista, livello) {
  return (lista || []).filter(function (r) {
    return eUsabile(r) && (livello === 'tutti' || r.livello === livello);
  });
}

/** Rende una frase confrontabile: spazi singoli, niente spazi ai bordi, maiuscole/minuscole ignorate.
    (Si ignora tutto il maiuscolo/minuscolo, non solo la prima lettera: i blocchi hanno una forma fissa,
    quindi "Ich" a metà frase deve valere come "ich" nelle frasi alternative.) */
function normalizzaFrase(testo) {
  return String(testo).normalize('NFC').trim().replace(/\s+/g, ' ').toLowerCase();
}

/** Controlla se i blocchi messi dal giocatore (lista di testi) formano la frase giusta
    o una delle `alternative` del record. Si confrontano le frasi complete. */
function controllaOrdine(blocchiScelti, record) {
  const costruita = normalizzaFrase(blocchiScelti.join(' '));
  const ammesse = [record.testo_de].concat(record.alternative || []);
  return ammesse.some(function (frase) { return normalizzaFrase(frase) === costruita; });
}

/** Dà un ordine mescolato degli indici 0..n-1 dei blocchi, diverso dall'ordine giusto
    (se i blocchi non sono tutti uguali). Restituisce una lista di indici. */
function ordineMescolato(blocchi, rng) {
  const indici = blocchi.map(function (_, i) { return i; });
  const originale = blocchi.join('\u0001');
  const comeTesto = function (ord) { return ord.map(function (i) { return blocchi[i]; }).join('\u0001'); };
  for (let tentativo = 0; tentativo < 20; tentativo++) {
    const prova = mescola(indici, rng);
    if (comeTesto(prova) !== originale) return prova;
  }
  // Ultima risorsa: rotazione di un posto (diversa dall'originale se i blocchi non sono tutti uguali).
  return indici.slice(1).concat(indici.slice(0, 1));
}

/** Una domanda di "Trova l'errore": le due frasi in ordine casuale. */
function opzioniErrore(record, rng) {
  const casuale = rng || Math.random;
  const opzioni = [
    { testo: record.frase_sbagliata, corretta: false },
    { testo: record.frase_corretta, corretta: true }
  ];
  return casuale() < 0.5 ? opzioni : [opzioni[1], opzioni[0]];
}

/** Contatori a zero. */
function nuovoProgresso() {
  const giochi = {};
  GIOCHI.forEach(function (g) { giochi[g] = { giusti: 0, sbagliati: 0, migliorSerie: 0 }; });
  return { versione: 1, livello: 'tutti', giochi: giochi };
}

/** Numero intero >= 0, altrimenti 0 (difesa contro dati rovinati nel localStorage). */
function interoSicuro(n) {
  return (typeof n === 'number' && isFinite(n) && n >= 0) ? Math.floor(n) : 0;
}

/** Costruisce un progresso valido partendo da un oggetto qualsiasi (letto dal browser). */
function sistemaProgresso(grezzo) {
  const p = nuovoProgresso();
  if (!grezzo || typeof grezzo !== 'object') return p;
  if (LIVELLI.indexOf(grezzo.livello) !== -1) p.livello = grezzo.livello;
  const g = grezzo.giochi && typeof grezzo.giochi === 'object' ? grezzo.giochi : {};
  GIOCHI.forEach(function (nome) {
    const v = g[nome] && typeof g[nome] === 'object' ? g[nome] : {};
    p.giochi[nome].giusti = interoSicuro(v.giusti);
    p.giochi[nome].sbagliati = interoSicuro(v.sbagliati);
    p.giochi[nome].migliorSerie = interoSicuro(v.migliorSerie);
  });
  return p;
}

/** Registra una risposta. `serie` = serie di risposte giuste di fila DOPO questa risposta. */
function registraRisposta(progresso, gioco, giusta, serie) {
  const c = progresso.giochi[gioco];
  if (giusta) c.giusti += 1; else c.sbagliati += 1;
  if (serie > c.migliorSerie) c.migliorSerie = serie;
  return progresso;
}

/** Totali per la home: risposte giuste, sbagliate, migliore serie tra i giochi. */
function riepilogo(progresso) {
  let giusti = 0, sbagliati = 0, migliorSerie = 0;
  GIOCHI.forEach(function (g) {
    const c = progresso.giochi[g];
    giusti += c.giusti;
    sbagliati += c.sbagliati;
    migliorSerie = Math.max(migliorSerie, c.migliorSerie);
  });
  return { giusti: giusti, sbagliati: sbagliati, migliorSerie: migliorSerie };
}

      btn.disabled = (n === 0);
      const c = progresso.giochi[g];
      document.getElementById('stat-' + g).textContent =
        'Giuste: ' + c.giusti + ' · Sbagliate: ' + c.sbagliati + ' · Serie migliore: ' + c.migliorSerie;
    });
    const tot = riepilogo(progresso);
    const risposte = tot.giusti + tot.sbagliati;
    document.getElementById('riepilogo').textContent = risposte === 0
      ? 'Non hai ancora risposto a nulla: scegli un gioco e inizia!'
      : 'Risposte giuste: ' + tot.giusti + ' su ' + risposte + ' (' + Math.round(100 * tot.giusti / risposte) +
        '%) · Serie migliore: ' + tot.migliorSerie;
    const nessunDato = GIOCHI.every(function (g) { return dati[g].length === 0; });
    avvisoDati.hidden = !nessunDato;
  }

  function creaSessione(gioco, alProssimo) {
    const sessione = { serie: 0 };
    const barra = el('p', { classe: 'barra-serie' });
    const corpo = el('div', { classe: 'corpo' });
    const messaggio = el('div', { classe: 'messaggio', attr: { role: 'status', 'aria-live': 'polite' } });
    const avanti = bottone('Avanti →', 'primario', alProssimo);
    avanti.hidden = true;
    const azioni = el('div', { classe: 'azioni' }, [avanti]);
    area.appendChild(barra);
    area.appendChild(corpo);
    area.appendChild(messaggio);
    area.appendChild(azioni);

    function aggiornaBarra() {
      barra.textContent = 'Serie: ' + sessione.serie + ' · Record: ' + progresso.giochi[gioco].migliorSerie;
    }
    aggiornaBarra();

    sessione.corpo = corpo;
    sessione.azioni = azioni;
    sessione.avanti = avanti;

    /** Prepara una nuova domanda: toglie messaggio e tasto Avanti. */
    sessione.nuovaDomanda = function () {
      svuota(messaggio);
      messaggio.className = 'messaggio';
      avanti.hidden = true;
    };
    /** Registra la risposta (giusta/sbagliata) e aggiorna serie e contatori. */
    sessione.registra = function (giusta) {
      sessione.serie = giusta ? sessione.serie + 1 : 0;
      registraRisposta(progresso, gioco, giusta, sessione.serie);
      salva();
      aggiornaBarra();
    };
    /** Mostra un messaggio: `tipo` = 'ok' | 'ko' | 'info'; `righe` = lista di {testo, lingua}. */
    sessione.messaggio = function (tipo, righe) {
      svuota(messaggio);
      messaggio.className = 'messaggio ' + tipo;
      righe.forEach(function (r) {
        messaggio.appendChild(el('p', { testo: r.testo, lingua: r.lingua, classe: r.classe }));
      });
    };
    /** Domanda conclusa: mostra Avanti e ci porta il focus. */
    sessione.mostraAvanti = function () {
      avanti.hidden = false;
      avanti.focus();
    };
    return sessione;
  }

  function giocoFrasi() {
    const pool = disponibili('frasi');
    mostraGioco('Rimetti in ordine');
    let corrente = null;
    let ordine = [];       // indici dei blocchi nell'ordine mescolato
    let scelti = [];       // indici dei blocchi scelti, nell'ordine in cui li ha messi il giocatore
    let chiusa = false;    // domanda conclusa (giusta o soluzione mostrata)
    let registrata = false; // la prima verifica conta una sola volta
    const sessione = creaSessione('frasi', prossima);

    function prossima() {
      corrente = scegliCasuale(pool, null, corrente);
      ordine = ordineMescolato(corrente.blocchi);
      scelti = [];
      chiusa = false;
      registrata = false;
      sessione.nuovaDomanda();
      disegna(null);
    }

    /** Ridisegna la domanda. `focus` = indice del blocco a cui dare il focus, 'controlla' o null. */
    function disegna(focus) {
      const corpo = sessione.corpo;
      svuota(corpo);
      corpo.appendChild(el('p', { classe: 'istruzione', testo: 'Tocca i blocchi nell’ordine giusto per formare la frase.' }));
      corpo.appendChild(el('p', { classe: 'suggerimento', testo: 'Suggerimento: ' + corrente.traduzione_it }));

      // frase costruita finora (toccando un blocco lo si toglie)
      const costruita = el('div', { classe: 'costruita', attr: { role: 'group', 'aria-label': 'La tua frase' } });
      if (scelti.length === 0) {
        costruita.appendChild(el('span', { classe: 'segnaposto', testo: 'La tua frase comparirà qui' }));
      }
      scelti.forEach(function (i) {
        costruita.appendChild(bottone(corrente.blocchi[i], 'blocco scelto', function () { togli(i); },
          { lang: 'de', 'data-i': String(i), 'aria-label': corrente.blocchi[i] + ' (tocca per toglierlo)' }));
      });
      corpo.appendChild(costruita);

      // blocchi ancora da usare
      const rimasti = el('div', { classe: 'rimasti', attr: { role: 'group', 'aria-label': 'Blocchi disponibili' } });
      ordine.forEach(function (i) {
        if (scelti.indexOf(i) === -1) {
          rimasti.appendChild(bottone(corrente.blocchi[i], 'blocco', function () { metti(i); },
            { lang: 'de', 'data-i': String(i) }));
        }
      });
      corpo.appendChild(rimasti);

      if (chiusa) {
        costruita.querySelectorAll('button').forEach(function (b) { b.disabled = true; });
        return;
      }

      const azioni = el('div', { classe: 'azioni' });
      const controlla = bottone('Controlla', 'primario', controllaClick, { id: 'btn-controlla' });
      azioni.appendChild(controlla);
      azioni.appendChild(bottone('Ricomincia', '', function () { scelti = []; sessione.messaggio('info', []); disegna(null); }));
      azioni.appendChild(bottone('Mostra la soluzione', '', mostraSoluzione));
      corpo.appendChild(azioni);

      if (focus === 'controlla') {
        controlla.focus();
      } else if (focus !== null && focus !== undefined) {
        const b = corpo.querySelector('button[data-i="' + focus + '"]');
        if (b) b.focus();
      }
    }

    function metti(i) {
      if (chiusa) return;
      scelti.push(i);
      // dopo aver scelto, il focus passa al prossimo blocco libero (o a "Controlla")
      const prossimo = ordine.find(function (k) { return scelti.indexOf(k) === -1; });
      disegna(prossimo === undefined ? 'controlla' : prossimo);
    }

    function togli(i) {
      if (chiusa) return;
      scelti = scelti.filter(function (k) { return k !== i; });
      disegna(i);
    }

    function controllaClick() {
      if (scelti.length < corrente.blocchi.length) {
        sessione.messaggio('info', [{ testo: 'Usa tutti i blocchi prima di controllare.' }]);
        return;
      }
      const testi = scelti.map(function (i) { return corrente.blocchi[i]; });
      if (controllaOrdine(testi, corrente)) {
        if (!registrata) sessione.registra(true);
        registrata = true;
        chiusa = true;
        sessione.messaggio('ok', [
          { testo: 'Giusto!', classe: 'esito' },
          { testo: testi.join(' '), lingua: 'de', classe: 'de-grande' },
          { testo: corrente.traduzione_it }
        ]);
        disegna(null);
        sessione.mostraAvanti();
      } else {
        if (!registrata) sessione.registra(false);
        registrata = true;
        sessione.messaggio('ko', [{ testo: 'Non è ancora giusto: riprova, oppure guarda la soluzione.', classe: 'esito' }]);
      }
    }

    function mostraSoluzione() {
      if (!registrata) sessione.registra(false);
      registrata = true;
      chiusa = true;
      sessione.messaggio('info', [
        { testo: 'Una soluzione possibile:' },
        { testo: corrente.testo_de, lingua: 'de', classe: 'de-grande' },
        { testo: corrente.traduzione_it }
      ]);
      disegna(null);
      sessione.mostraAvanti();
    }

    prossima();
  }

      const gruppo = el('div', { classe: 'due-opzioni', attr: { role: 'group', 'aria-label': 'Le due frasi' } });
      opzioniErrore(corrente).forEach(function (op) {
        const b = bottone(op.testo, 'opzione', function () { rispondi(op, gruppo); }, { lang: 'de' });
        b._corretta = op.corretta;
        gruppo.appendChild(b);
      });
      sessione.corpo.appendChild(gruppo);
      gruppo.firstChild.focus();
    }

    function rispondi(scelta, gruppo) {
      gruppo.querySelectorAll('button').forEach(function (b) {
        b.disabled = true;
        if (b._corretta) {
          b.classList.add('giusto');
          b.textContent = '✓ ' + b.textContent;
        } else if (b.textContent === scelta.testo) {
          b.classList.add('sbagliato');
          b.textContent = '✗ ' + b.textContent;
        }
      });
      sessione.registra(scelta.corretta);
      sessione.messaggio(scelta.corretta ? 'ok' : 'ko', [
        { testo: scelta.corretta ? 'Giusto!' : 'Non proprio.', classe: 'esito' },
        { testo: 'Frase corretta: ' + corrente.frase_corretta, lingua: 'de', classe: 'de-grande' },
        { testo: corrente.spiegazione_it }
      ]);
      sessione.mostraAvanti();
    }

    prossima();
  }

  });

}

# SCHEMA – come è fatto un contenuto di Langue

Ogni contenuto (parola, frase, errore, consiglio) è un "record": un elenco di campi con nome e valore.
Le regole complete sono in `schema/contenuto.schema.json`; esempi giusti in `schema/esempi_validi.json`, esempi sbagliati (da rifiutare) in `schema/esempi_errati.json`.
In un record si possono usare **solo** i campi elencati qui sotto: un campo in più rende il record non valido.

## Campi comuni (tutti i tipi)

| Campo | Cosa è | Esempio |
|---|---|---|
| id | nome unico del record | `esempio-articolo-1` |
| tipo | `articolo`, `frase`, `errore`, `consiglio` | `articolo` |
| lingua | sempre `de` | `de` |
| livello | `A1`, `A2`, `B1`, `altro` | `A1` |
| stato | `bozza`, `verificato`, `da_verificare` | `bozza` |
| origine | chi ha creato il record | `importato`, `agente T07` |
| fonte (facoltativo) | da dove viene l'informazione | – |
| note (facoltativo) | appunto breve per chi rilegge | `articolo da controllare` |

## I quattro tipi

**articolo** – un nome con il suo der/die/das.
- `parola`: `Tisch` · `articolo`: `der` · `traduzione_it`: `tavolo`

**frase** – una frase tedesca da rimettere in ordine.
- `testo_de`: `Ich lerne heute Deutsch`
- `traduzione_it`: `Oggi studio tedesco`
- `blocchi`: da 4 a 9 pezzi nell'ordine giusto: `["Ich", "lerne", "heute", "Deutsch"]`. Uniti da uno spazio devono dare **esattamente** `testo_de`.
- `alternative` (facoltativo): altri ordini corretti, come frasi complete: `["Heute lerne ich Deutsch"]`

**errore** – un errore tipico di chi parla italiano.
- `frase_sbagliata`: `Ich habe ein Hund.` · `frase_corretta`: `Ich habe einen Hund.`
- `categoria`: `articolo`, `ordine_verbo`, `genere`, `falsi_amici`, `preposizioni`, `altro`
- `spiegazione_it`: spiegazione in italiano (almeno 20 caratteri)

**consiglio** – un consiglio di film, serie, libri o brani (più avanti).
- `titolo`: titolo reale dell'opera
- `genere`: `thriller_crime`, `commedia`, `fantasy_fantascienza`, `musica_pop_rock`
- `supporto`: `film`, `serie`, `libro`, `brano`
- `perche_adatto_it`: perché è adatto a chi impara
- `parole_utili`: da 3 a 5 parole tedesche: `["die Polizei", "der Mord", "die Leiche"]`

## Come si segna un contenuto

- Contenuto nuovo creato da un agente: `"origine": "agente T0X"` e `"stato": "bozza"`.
- Contenuto importato dai vecchi dati: `"origine": "importato"`.
- Dubbio su tedesco o traduzione: `"stato": "da_verificare"` e una riga in `note` che spiega il dubbio.
- **Per segnarlo come verificato**: dopo averlo riletto, cambia a mano `"stato": "bozza"` in `"stato": "verificato"`. Non cambiare altro.

## Riepilogo delle regole

| Regola | Valore |
|---|---|
| Campi obbligatori | tutti, tranne `fonte`, `note`, `alternative` |
| Campi ammessi | solo quelli del proprio tipo + comuni |
| `lingua` | solo `de` |
| `blocchi` (frase) | 4–9 elementi, uniti = `testo_de` |
| `spiegazione_it` (errore) | almeno 20 caratteri |
| `parole_utili` (consiglio) | 3–5 elementi |
| `alternative` (frase) | se presente, 1–5 elementi |
| Testi | non vuoti, con lunghezza massima |

## Confronto
| Idea | Proposta da | Valore | Difficoltà |
|------|-------------|--------|------------|
| Mostrare l'articolo corretto sotto la parola grande dopo ogni risposta nel gioco articoli | agente T18-idee-a | Alto | MEDIO |
| Tracciamento per-item + modalità "Ripassa i tuoi errori" (debito per gioco) | agente T18-idee-b | Alto | MEDIO |
| "Ripasso degli sbagli" (lista dei debiti con rimozione al successo) | agente T18-idee-c | Alto | MEDIO |
| Aggiungere percentuale di risposte giuste alle statistiche della home | agente T18-idee-a | Medio | FACILE |
| Export/Import dei progressi tramite file JSON | agente T18-idee-b | Medio | FACILE |

## Scelta
**Ripasso degli sbagli (lista dei debiti)** – l'idea proposta da entrambi gli agenti T18-idee-b e T18-idee-c. Si aggiunge una chiave separata in `localStorage` (`langue_debiti_v1`) che registra, per ogni gioco, gli item su cui l'utente ha risposto male o ha rinunciato. Una nuova scheda in home "Ripasso (n)" apre una sessione che presenta solo quei debiti, ordinati per frequenza di errore; ogni risposta corretta rimuove l'item dal debito. Il tutto si realizza modificando solo `web/app.js`, `web/style.css` e `web/index.html`, senza aggiungere contenuti tedeschi.

## Brief per chi la realizza
1. **app.js – nuove funzioni di debito**  
   - Aggiungere le costanti `CHIAVE_DEBITI = 'langue_debiti_v1'`.  
   - Funzioni: `caricaDebiti(storage)`, `salvaDebiti(storage, debito)`, `aggiungiDebito(gioco, id)`, `rimuoviDebito(gioco, id)`, `getDebitiPerGioco(gioco)` (restituisce array di `{id, volte}` ordinato per `volte` decrescente, max 50 per gioco).  
   - In `avviaSito` leggere i debiti insieme ai progressi.  

2. **app.js – registrazione dei debiti**  
   - Nella funzione `rispondi` di `giocoArticoli`, dopo `sessione.registra(giusta)`, se `!giusta` chiamare `aggiungiDebito('articoli', corrente.id)`.  
   - Nella funzione `controllaClick` di `giocoFrasi`, se la risposta è sbagliata (il primo `else`), chiamare `aggiungiDebito('frasi', corrente.id)`.  
   - Nella funzione `mostraSoluzione` di `giocoFrasi`, sempre chiamare `aggiungiDebito('frasi', corrente.id)` (contare la rinuncia come errore).  
   - Nella funzione `rispondi` di `giocoErrori`, se `!scelta.corretta`, chiamare `aggiungiDebito('errori', corrente.id)`.  
   - Nella funzione `mostraSoluzione` di `giocoErrori` (non presente, ma analogamente se aggiunta) vale lo stesso principio.  

3. **app.js – sessione di ripasso**  
   - Creare funzione `creaSessioneRipasso(gioco, alProssimo)` simile a `creaSessione` ma che:  
     * prende il pool da `getDebitiPerGioco(gioco)` (array di oggetti con `id` e `voli`);  
     * per ogni `id` recupera il record originale da `window.LANGUE_DATA[gioco]` tramite ricerca per `id`;  
     * mostra la domanda usando i renderer esistenti (es. per articoli mostra la parola grande e i bottoni articolo).  
   - Quando l'utente risponde correttamente, chiamare `rimuoviDebito(gioco, id)` e aggiornare il conteggio in home.  
   - Se il pool di debiti è vuoto, mostrare un messaggio "Nessun debito da ripassare" e tornare alla home.  

4. **app.js – avvio ripasso dalla home**  
   - Aggiungere funzione `avviaRipasso(gioco)` che chiama `creaSessioneRipasso(gioco, prossimaRipasso)` dove `prossimaRipasso` richiama nuovamente `avviaRipasso` per lo stesso gioco finché ci sono debiti, oppure torna alla home quando finiti.  
   - Collegare il nuovo bottone della home (vedi punto 5) a `avviaRipasso` con il gioco corrispondente (`articoli`, `frasi` o `errori`).  

5. **index.html – nuova scheda in home**  
   - Dentro `<ul class="giochi">` aggiungere un nuovo `<li>` con id `gioca-ripasso` e classe `scheda`.  
   - All'interno: `<span class="scheda-titolo">Ripasso</span>`, `<span class="scheda-desc">Rispondi nuovamente agli errori</span>`, `<span class="scheda-conta" id="conta-ripasso"></span>`, `<span class="scheda-stat" id="stat-ripasso"></span>`.  
   - Aggiungere listener: `document.getElementById('gioca-ripasso').addEventListener('click', function() { avviaRipasso(progresso.livello); });` (il livello è già nel progresso, ma la sessione di ripasso ignora il filtro livello e usa tutti i debiti indipendentemente dal livello).  

6. **style.css – eventuale aggiustamento**  
   - Nessuna modifica obbligatoria; la nuova scheda eredita già le regole `.scheda`. Se si desidera evidenziarla, si può aggiungere una regola specifica per `#gioca-ripasso` (facoltativo).  

7. **Cosa NON modificare**  
   - Non toccare la struttura dei giochi esistenti (`giocoArticoli`, `giocoFrasi`, `giocoErrori`) oltre alle chiamate di aggiunta/rimozione debito.  
   - Non alterare l'oggetto `progresso` né le chiavi di `localStorage` usate per i progressi (`langue_progress_v1`).  
   - Non introdurre nuovi contenuti tedeschi; i debiti riutilizzano i record già presenti in `data.js`.  

8. **Come verificare**  
   - Dopo una risposta sbagliata (o dopo aver cliccato "Mostra la soluzione") verificare che il contatore accanto alla scheda "Ripasso" aumenti di 1.  
   - Cliccare su "Ripasso (n)" e confermare che la sessione presenti l'item appena sbagliato.  
   - Rispondere correttamente in quella sessione: l'item deve scomparire dal debito e il contatore diminuire.  
   - Continuare a sbagliare lo stesso item: il contatore delle volte (`voli`) deve aumentare e l'item rimanere più in alto nella lista.  
   - Assicurarsi che il numero massimo di debiti per gioco non superi 50 (più vecchi vengono scartati quando si aggiunge un nuovo debito oltre il limite).  

## Idee scartate e perche'
- **Mostrare l'articolo corretto dopo risposta** (idea-a #2): utile ma impatto inferiore rispetto a un sistema di ripasso mirato; inoltre richiede solo modifiche di stile e testo, mentre il ripasso offre un beneficio di lungo periodo.  
- **Tracciamento per-item + modalità "Ripassa i tuoi errori"** (idea-b #1): praticamente identico alla scelta, ma la proposta dell'agente c è più precisa sulla rimozione al successo e sul limite di 50 item; abbiamo adottato quella versione perché è più completa e già testata mentalmente.  
- **Aggiungere percentuale di risposte giuste alle statistiche** (idea-a #5): facile da fare, ma fornisce solo una metrica globale senza guidare l'utente su cosa ripassare.  
- **Export/Import dei progressi** (idea-b #4): utile per la portabilità dei dati, ma non influisce direttamente sull'apprendimento quotidiano; inoltre richiederebbe nuovi pulsanti e gestione di file, mentre il ripasso si ottiene con modifiche più locali e immediato beneficio.  
- Altre idee (navigazione tastiera, hint, service worker, obiettivo giornaliero, ecc.) sono state scartate perché o richiedono nuovi file fuori dalle allowed paths (es. service worker, manifest, test) o hanno un impatto didattico inferiore rispetto al ripasso mirato degli errori.
